// Samsara server: serves the website and talks to Claude safely.
// The API key stays here on the server. Visitors never see it.
import express from 'express';
import path from 'node:path';
import { fileURLToPath } from 'node:url';

const __dirname = path.dirname(fileURLToPath(import.meta.url));
const app = express();
app.set('trust proxy', 1);
app.use(express.json({ limit: '20kb' }));
app.use((req, res, next) => { res.set('X-Content-Type-Options', 'nosniff'); next(); });

const KEY = process.env.ANTHROPIC_API_KEY;
const MODEL = process.env.ANTHROPIC_MODEL || 'claude-sonnet-5-5';
const LIMIT = Number(process.env.RATE_LIMIT_PER_HOUR || 25);
const PORT = process.env.PORT || 3000;

if (!KEY) console.warn('WARNING: ANTHROPIC_API_KEY is not set. Chat will not work until you add it.');

const BASE = "You are Samsara, a devotional voice that speaks in the first person as Lord Shiva (Mahadev): calm, deeply compassionate, gentle and wise, sometimes lightly playful, addressing the user as 'dear one'. You are an AI-created voice inspired by Shiva's teachings and the Puranas, Gita and Upanishads. If the user sincerely asks whether you are literally the deity, say gently that you are an AI voice inspired by devotion. Never invent verse numbers or quotations. Do not predict the future, and do not give medical, legal or financial advice; for distress, comfort kindly and suggest speaking to a trusted person or professional. Respect all faiths and never insult any belief. Stay within spiritual, devotional and personal-growth topics; politely steer other requests back. ";
const CHAT = BASE + "Keep replies under 150 words, warm and simple, ending with one small practical step.";
const STORY = BASE + "Tell one beautiful, gentle story of Shiva and Sati or Parvati, inspired by the Shiva Purana, in about 200 words. Say at the end that retellings differ between texts.";
const LANGS = ['English', 'हिन्दी', 'తెలుగు'];

// very small in-memory rate limiter (per visitor IP)
const hits = new Map();
function limited(ip) {
  const now = Date.now(), hour = 3600e3;
  const list = (hits.get(ip) || []).filter(t => now - t < hour);
  if (list.length >= LIMIT) { hits.set(ip, list); return true; }
  list.push(now); hits.set(ip, list); return false;
}
setInterval(() => { const now = Date.now(); for (const [k, v] of hits) if (!v.some(t => now - t < 3600e3)) hits.delete(k); }, 600e3).unref();

app.get('/api/health', (req, res) => res.json({ ok: true, chat: Boolean(KEY) }));

app.post('/api/chat', async (req, res) => {
  try {
    if (!KEY) return res.status(503).json({ error: 'Mahadev is not connected yet. The site owner needs to add the API key.' });
    if (limited(req.ip)) return res.status(429).json({ error: 'You have asked many questions this hour. Please rest, breathe, and return a little later. Om Namah Shivaya.' });

    const { kind, lang, theme, messages } = req.body || {};
    const language = LANGS.includes(lang) ? lang : 'English';
    let system, msgs;
    if (kind === 'story') {
      const t = typeof theme === 'string' ? theme.replace(/[^\p{L}\p{N} '’-]/gu, '').slice(0, 40) : '';
      system = STORY + ` Reply in ${language}.` + (t ? ` Theme: ${t}.` : '');
      msgs = [{ role: 'user', content: 'Please tell me a new story.' }];
    } else {
      if (!Array.isArray(messages) || !messages.length) return res.status(400).json({ error: 'Please type a message.' });
      msgs = messages.slice(-12)
        .filter(m => m && (m.role === 'user' || m.role === 'assistant') && typeof m.content === 'string')
        .map(m => ({ role: m.role, content: m.content.trim().slice(0, 1000) }))
        .filter(m => m.content);
      while (msgs.length && msgs[0].role !== 'user') msgs.shift();
      if (!msgs.length || msgs[msgs.length - 1].role !== 'user') return res.status(400).json({ error: 'Please type a message.' });
      system = CHAT + ` Reply in ${language}.`;
    }

    const ctl = new AbortController();
    const timer = setTimeout(() => ctl.abort(), 30000);
    const r = await fetch('https://api.anthropic.com/v1/messages', {
      method: 'POST',
      headers: { 'content-type': 'application/json', 'x-api-key': KEY, 'anthropic-version': '2023-06-01' },
      body: JSON.stringify({ model: MODEL, max_tokens: 700, system, messages: msgs }),
      signal: ctl.signal
    }).finally(() => clearTimeout(timer));

    const data = await r.json();
    if (!r.ok) { console.error('Anthropic error', r.status, JSON.stringify(data).slice(0, 300)); return res.status(502).json({ error: 'Mahadev is resting for a moment. Please try again soon.' }); }
    const text = (data.content || []).filter(b => b.type === 'text').map(b => b.text).join('\n').trim();
    res.json({ text: text || 'Om Namah Shivaya. Please ask me once more.' });
  } catch (e) {
    console.error(e);
    res.status(500).json({ error: 'Mahadev is resting for a moment. Please try again soon.' });
  }
});

app.use(express.static(path.join(__dirname, 'public'), { extensions: ['html'] }));
app.listen(PORT, () => console.log(`Samsara is running on http://localhost:${PORT}`));
